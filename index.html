import pygame import sys import random import math # Initialize Engine pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("THE COAL PIT: REAL-TIME EVOLUTION") clock = pygame.time.Clock() # Color Palettes (Retro Pixels) COLOR_BG = (10, 10, 14) COLOR_WALL = (45, 42, 50) COLOR_TRACK = (90, 85, 95) COLOR_COAL = (25, 25, 30) # Game Variables TILE_SIZE = 40 GRID_W = WIDTH // TILE_SIZE GRID_H = HEIGHT // TILE_SIZE class GameState: def __init__(self): self.level = 1 self.coal_collected = 0 self.coal_needed = 3 self.game_over = False self.game_won = False self.generate_level() def generate_level(self): self.coal_collected = 0 # 1. Procedural Mine Layout Generation (Tunnels rather than Mazes) self.grid = [[1 for _ in range(GRID_W)] for _ in range(GRID_H)] # Carve Main Railway Tracks loop for x in range(2, GRID_W - 2): self.grid[GRID_H // 2][x] = 0 for y in range(2, GRID_H - 2): self.grid[y][GRID_W // 2] = 0 # Carve out mining pockets self.pockets = [(3, 3), (GRID_W - 4, 3), (3, GRID_H - 4), (GRID_W - 4, GRID_H - 4)] for px, py in self.pockets: for dx in range(-1, 2): for dy in range(-1, 2): self.grid[py + dy][px + dx] = 0 # Place Objects self.player_x, self.player_y = 2, GRID_H // 2 self.cart_x, self.cart_y = 3, GRID_H // 2 self.elevator_x, self.elevator_y = GRID_W - 3, GRID_H // 2 # Spawn Coal Nodes self.coal_nodes = [{"x": p, "y": p, "active": True} for p in self.pockets[:3]] # Configure Evolving Enemy (Real-time tracking attributes) self.monster = Monster(GRID_W - 4, GRID_H - 4, self.level) class Monster: def __init__(self, x, y, level): self.x = x self.y = y self.stage = level - 1 self.anim_frame = 0 # Time-based movement speeds (in milliseconds) # Level 1 = 600ms per step. Higher levels speed up drastically. self.move_speed_ms = max(200, 600 - (level * 80)) self.last_move_time = pygame.time.get_ticks() def animate(self): # Frame counter loops continuously for heavy step/breathing effects self.anim_frame = (self.anim_frame + 0.1) % 4 def update_position_realtime(self, current_time, p_x, p_y, grid): # Check if enough time has passed on the engine clock to take a step if current_time - self.last_move_time < self.move_speed_ms: return self.last_move_time = current_time # Default target vector directions dx = int(math.copysign(1, p_x - self.x)) if p_x != self.x else 0 dy = int(math.copysign(1, p_y - self.y)) if p_y != self.y else 0 # 🟢 Real-Time Mutation AI Decision Tree if self.stage == 0: # Stage 0: Completely blind wanderer. Chooses a random open direction card. valid_moves = [] for ox, oy in [(0,1), (0,-1), (1,0), (-1,0)]: if grid[self.y + oy][self.x + ox] == 0: valid_moves.append((ox, oy)) if valid_moves: dx, dy = random.choice(valid_moves) elif self.stage == 1: # Stage 1: Weak Tracking. 35% chance to wander stupidly, otherwise hunts vectors. if random.random() < 0.35: valid_moves = [] for ox, oy in [(0,1), (0,-1), (1,0), (-1,0)]: if grid[self.y + oy][self.x + ox] == 0: valid_moves.append((ox, oy)) if valid_moves: dx, dy = random.choice(valid_moves) # Execute physical grid movement step if no walls block path if grid[self.y + dy][self.x + dx] == 0: self.x += dx self.y += dy elif grid[self.y][self.x + dx] == 0: # Smart sliding slide checks self.x += dx elif grid[self.y + dy][self.x] == 0: self.y += dy # Instantiation game = GameState() anim_tick = 0 def draw_mine_environment(): for y in range(GRID_H): for x in range(GRID_W): rect = pygame.Rect(x * TILE_SIZE, y * TILE_SIZE, TILE_SIZE, TILE_SIZE) if game.grid[y][x] == 1: pygame.draw.rect(screen, COLOR_WALL, rect) pygame.draw.rect(screen, (20, 18, 22), rect, 2) else: pygame.draw.rect(screen, (18, 16, 15), rect) # Draw Cross-Ties Railway Track Networks for x in range(2, GRID_W - 2): y = GRID_H // 2 pygame.draw.line(screen, COLOR_TRACK, (x*TILE_SIZE, y*TILE_SIZE + 15), ((x+1)*TILE_SIZE, y*TILE_SIZE + 15), 2) pygame.draw.line(screen, COLOR_TRACK, (x*TILE_SIZE, y*TILE_SIZE + 25), ((x+1)*TILE_SIZE, y*TILE_SIZE + 25), 2) for i in range(0, TILE_SIZE, 10): pygame.draw.line(screen, (60, 50, 40), (x*TILE_SIZE + i, y*TILE_SIZE + 10), (x*TILE_SIZE + i, y*TILE_SIZE + 30), 3) def draw_entities(): global anim_tick anim_tick += 0.1 # 1. Draw Coal Nodes for coal in game.coal_nodes: if coal["active"]: cx, cy = coal["x"] * TILE_SIZE + 20, coal["y"] * TILE_SIZE + 20 pygame.draw.circle(screen, COLOR_COAL, (cx, cy), 12) pygame.draw.circle(screen, (50, 55, 60), (cx - 3, cy - 3), 4) # 2. Draw Mine Cart mx, my = game.cart_x * TILE_SIZE + 4, game.cart_y * TILE_SIZE + 10 pygame.draw.rect(screen, (70, 75, 85), (mx, my, 32, 22)) pygame.draw.circle(screen, (20, 20, 20), (mx + 6, my + 24), 5) pygame.draw.circle(screen, (20, 20, 20), (mx + 26, my + 24), 5) if game.coal_collected > 0: pygame.draw.ellipse(screen, COLOR_COAL, (mx + 4, my - 4, 24, 12)) # 3. Draw Elevator Shaft Gate Area ex, ey = game.elevator_x * TILE_SIZE, game.elevator_y * TILE_SIZE pygame.draw.rect(screen, (100, 100, 120), (ex, ey, TILE_SIZE, TILE_SIZE), 4) for i in range(0, TILE_SIZE, 8): pygame.draw.line(screen, (140, 30, 30), (ex + i, ey), (ex + i, ey + TILE_SIZE), 1) # 4. Draw Animated Player Sprite px, py = game.player_x * TILE_SIZE + 10, game.player_y * TILE_SIZE + 5 bounce = int(math.sin(anim_tick) * 2) pygame.draw.circle(screen, (240, 240, 255), (px + 10, py + 8 + bounce), 7) pygame.draw.rect(screen, (40, 80, 150), (px + 4, py + 16, 12, 18)) pygame.draw.rect(screen, (255, 200, 0), (px + 6, py + 2 + bounce), 8, 3) # 5. Draw Animated Cobb Monster Sprite (Always processing animations) game.monster.animate() m_shift = int(math.sin(game.monster.anim_frame) * 4) mon_x = game.monster.x * TILE_SIZE mon_y = game.monster.y * TILE_SIZE # Monster body core pygame.draw.ellipse(screen, (130, 20, 20), (mon_x + 4, mon_y + 2, 32, 36)) # Mutated Grinning Jaw structure pygame.draw.rect(screen, (240, 240, 240), (mon_x + 8, mon_y + 16 + m_shift, 24, 6)) pygame.draw.rect(screen, (0, 0, 0), (mon_x + 12, mon_y + 16 + m_shift, 2, 6)) pygame.draw.rect(screen, (0, 0, 0), (mon_x + 18, mon_y + 16 + m_shift, 2, 6)) # Reaching asymmetric arm frames pygame.draw.line(screen, (100, 10, 10), (mon_x + 4, mon_y + 12), (mon_x - 6, mon_y + 24 + m_shift), 4) pygame.draw.line(screen, (100, 10, 10), (mon_x + 36, mon_y + 12), (mon_x + 46, mon_y + 20 - m_shift), 4) def apply_flashlight_vignette(): mask = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA) mask.fill((3, 3, 6, 252)) p_screen_x = game.player_x * TILE_SIZE + 20 p_screen_y = game.player_y * TILE_SIZE + 20 for radius in range(160, 40, -4): alpha = int(255 * (radius / 160)) pygame.draw.circle(mask, (0, 0, 0, alpha), (p_screen_x, p_screen_y), radius) screen.blit(mask, (0, 0)) # --- MAIN CORE REAL-TIME GAME LOOP --- while True: current_time = pygame.time.get_ticks() screen.fill(COLOR_BG) # 1. Continuous Independent Monster Behavior Update Thread if not game.game_over: game.monster.update_position_realtime(current_time, game.player_x, game.player_y, game.grid) # 2. Player Input Event Catchers for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() elif event.type == pygame.KEYDOWN and not game.game_over: dx, dy = 0, 0 if event.key in (pygame.K_LEFT, pygame.K_a): dx = -1 if event.key in (pygame.K_RIGHT, pygame.K_d): dx = 1 if event.key in (pygame.K_UP, pygame.K_w): dy = -1 if event.key in (pygame.K_DOWN, pygame.K_s): dy = 1 new_px = game.player_x + dx new_py = game.player_y + dy if game.grid[new_py][new_px] == 0: # Track pushing handler if new_px == game.cart_x and new_py == game.cart_y: if game.grid[game.cart_y + dy][game.cart_x + dx] == 0: game.cart_x += dx game.cart_y += dy game.player_x = new_px game.player_y = new_py else: game.player_x = new_px game.player_y = new_py # Coal interaction for coal in game.coal_nodes: if coal["active"] and game.player_x == coal["x"] and game.player_y == coal["y"]: coal["active"] = False game.coal_collected += 1 # 3. Game State Rule Checks if game.player_x == game.monster.x and game.player_y == game.monster.y: game.game_over = True if game.cart_x == game.elevator_x and game.cart_y == game.elevator_y and game.coal_collected >= game.coal_needed: game.level += 1 game.generate_level() # 4. Draw Frame Layers draw_mine_environment() draw_entities() apply_flashlight_vignette() # Render UI Layout Text Overlay font = pygame.font.SysFont("Courier New", 18, bold=True) ui_text = font.render(f"MINE DEPTH: B{game.level} | COAL IN CART: {game.coal_collected}/{game.coal_needed}", True, (255, 255, 255)) screen.blit(ui_text, (20, 20)) if game.game_over: loss_text = font.render("💀 COBB CONSUMED YOU. RE-RUN THE SCRIPT TO RESTART.", True, (255, 50, 50)) screen.blit(loss_text, (WIDTH//2 - 250, HEIGHT//2)) pygame.display.flip() clock.tick(60) # Limits system load rendering loops to a smooth 60fps
