import pygame import random # Initialize Pygame pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("8-Bit FNAF 4 Fan Game") clock = pygame.time.Clock() # Colors & Game States BLACK = (0, 0, 0) WHITE = (255, 255, 255) rooms = ["BEDROOM", "LEFT_HALL", "RIGHT_HALL", "CLOSET"] current_room = "BEDROOM" # Monster Variables monster_pos = 0 # 0: Far, 5: At Door (Jumpscare) monster_timer = pygame.USEREVENT + 1 pygame.time.set_timer(monster_timer, 5000) # Monster moves every 5 seconds running = True while running: # 1. Input Handling keys = pygame.key.get_pressed() flashlight = keys[pygame.K_f] # Flashlight (F) door_held = keys[pygame.K_SPACE] # Close Door (Space) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False # Room Navigation (WASD) if event.type == pygame.KEYDOWN: if event.key == pygame.K_a: current_room = "LEFT_HALL" if event.key == pygame.K_d: current_room = "RIGHT_HALL" if event.key == pygame.K_w: current_room = "CLOSET" if event.key == pygame.K_s: current_room = "BEDROOM" # Monster AI Logic if event.type == monster_timer: if not (current_room == "LEFT_HALL" and door_held): monster_pos += 1 if monster_pos >= 5: print("JUMPSCARE! Game Over.") running = False # 2. Rendering screen.fill(BLACK) # Draw Background (Placeholder text for your 8-bit sprites) font = pygame.font.SysFont("arial", 32) room_text = font.render(f"Location: {current_room}", True, WHITE) screen.blit(room_text, (20, 20)) if flashlight: pygame.draw.circle(screen, (255, 255, 200), (WIDTH//2, HEIGHT//2), 200) # Light effect if current_room == "LEFT_HALL" and monster_pos > 0: # If monster is flashed in hall, it retreats monster_pos = 0 print("Monster retreated!") if door_held and (current_room == "LEFT_HALL" or current_room == "RIGHT_HALL"): pygame.draw.rect(screen, (50, 50, 50), (100, 50, 600, 500)) # Door visual pygame.display.flip() clock.tick(60) pygame.quit()
